## Assinatura de governança manualmente
openssl smime -sign -in governance/governance.xml -text -out governance/governance_signed.p7s -signer certificates/permissions/permissions_ca_cert.pem -inkey certificates/permissions/permissions_ca_private_key.pem

## Geração de CSR (Certificate Signing Request)
openssl req -config identity_ca_openssl.cnf -new -key bookingPlatformDP_private_key.pem -out bookingPlatformDP.csr

## Verificação da assinatura de governança
openssl smime -verify -in governance/governance_signed.p7s -content governance/governance.xml -CAfile certificates/identity2/identity_ca_cert.pem
